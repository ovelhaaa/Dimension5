cmake_minimum_required(VERSION 3.16)
project(Dimension5 VERSION 0.1.0 LANGUAGES C)

option(DIMENSION5_ENABLE_ASAN "Enable AddressSanitizer for host builds" OFF)
option(DIMENSION5_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer for host builds" OFF)
option(DIMENSION5_ENABLE_FAST_MATH "Enable fast-math optimizations explicitly" OFF)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(dimension5_core STATIC src/dimension_dsp.c)
target_include_directories(dimension5_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(NOT MSVC)
  target_compile_options(dimension5_core PRIVATE -Wall -Wextra -Wshadow -Wdouble-promotion -Werror)
  if(DIMENSION5_ENABLE_FAST_MATH)
    target_compile_options(dimension5_core PRIVATE -ffast-math)
  endif()
endif()
target_link_libraries(dimension5_core PUBLIC m)

function(dimension5_apply_host_options target)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Wdouble-promotion -Werror)
    set(_san_flags "")
    if(DIMENSION5_ENABLE_ASAN)
      list(APPEND _san_flags -fsanitize=address)
    endif()
    if(DIMENSION5_ENABLE_UBSAN)
      list(APPEND _san_flags -fsanitize=undefined)
    endif()
    if(_san_flags)
      target_compile_options(${target} PRIVATE ${_san_flags})
      target_link_options(${target} PRIVATE ${_san_flags})
    endif()
  endif()
endfunction()

enable_testing()
add_executable(dimension5_tests tests/test_dimension_core.c)
target_link_libraries(dimension5_tests PRIVATE dimension5_core)
dimension5_apply_host_options(dimension5_tests)
add_test(NAME dimension5_tests COMMAND dimension5_tests)

add_executable(dimension5_offline_render tools/dimension5_offline_render.c)
target_link_libraries(dimension5_offline_render PRIVATE dimension5_core)
dimension5_apply_host_options(dimension5_offline_render)
